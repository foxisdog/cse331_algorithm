\begin{algorithm}
\caption{Comb Sort}
\begin{algorithmic}
\State $shrink \gets 1.3$
\State $is\_swap \gets \textbf{true}$
\State $gap \gets n$
\While{$is\_swap \textbf{ or } gap \neq 1$}
    \State $is\_swap \gets \textbf{false}$
    \State $gap \gets \lfloor gap / shrink \rfloor$
    \If{$gap < 1$}
        \State $gap \gets 1$
    \EndIf
    \For{$i \gets 0$ to $n - gap - 1$}
        \If{$arr[i] > arr[i+gap]$}
            \State Swap $arr[i]$ and $arr[i+gap]$
            \State $is\_swap \gets \textbf{true}$
        \EndIf
    \EndFor
\EndWhile
\end{algorithmic}
\end{algorithm}


\begin{algorithm}
\caption{Intro Sort}
\begin{algorithmic}
\Procedure{IntroSort}{$arr, n, max\_depth, threshold, recur, gen$}
    \If{$n \leq 1$}
        \State \Return
    \EndIf
    \If{$recur \geq max\_depth$}
        \State Call HeapSort($arr, n$)
        \State \Return
    \EndIf
    \If{$n \leq threshold$}
        \State Call InsertionSort($arr, n$)
        \State \Return
    \EndIf
    \State Select pivot randomly
    \State Partition array around pivot
    \State Recursively sort left and right partitions
\EndProcedure
\end{algorithmic}
\end{algorithm}


\begin{algorithm}
\caption{Library Sort}
\begin{algorithmic}
\Procedure{LibrarySort}{$arr, n$}
    \State Initialize temporary array $tmp$ and gap array $is\_gap$
    \State Copy first element of $arr$ to $tmp$
    \For{$i \gets 1$ to $\lfloor \log_2(n) \rfloor + 1$}
        \State Rebalance $tmp$ and $is\_gap$
        \For{$j \gets 0$ to $n - 1$}
            \State Find position using BinarySearch
            \State Insert element into $tmp$ at the found position
        \EndFor
    \EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}


\begin{algorithm}
\caption{Shaker Sort}
\begin{algorithmic}
\Procedure{ShakerSort}{$arr, n$}
    \State $is\_swap \gets \textbf{true}$
    \State $ub \gets n - 1$
    \State $lb \gets 0$
    \While{$is\_swap$}
        \State $is\_swap \gets \textbf{false}$
        \Comment{Move right}
        \For{$position \gets lb$ to $ub - 1$}
            \If{$arr[position] > arr[position + 1]$}
                \State Swap $arr[position]$ and $arr[position + 1]$
                \State $last \gets position$
                \State $is\_swap \gets \textbf{true}$
            \EndIf
        \EndFor
        \State $ub \gets last$
        \Comment{Move left}
        \For{$position \gets ub$ to $lb + 1$}
            \If{$arr[position - 1] > arr[position]$}
                \State Swap $arr[position]$ and $arr[position - 1]$
                \State $last \gets position$
                \State $is\_swap \gets \textbf{true}$
            \EndIf
        \EndFor
        \State $lb \gets last$
    \EndWhile
\EndProcedure
\end{algorithmic}
\end{algorithm}


\begin{algorithm}
\caption{Tim Sort}
\begin{algorithmic}
\Procedure{TimSort}{$arr, n$}
    \State $minirun \gets 32$
    \State Initialize $runs$ as an empty list
    \Comment{Create runs}
    \State $i \gets 0$
    \While{$i < n$}
        \State $runend \gets$ Call MakerRun($arr, n, i, i + minirun$)
        \State Append $(i, runend)$ to $runs$
        \State $i \gets runend + 1$
    \EndWhile
    \Comment{Merge runs}
    \While{$runs.size() > 1$}
        \State $a \gets$ RunSize($runs[runs.size() - 1]$)
        \State $b \gets$ RunSize($runs[runs.size() - 2]$)
        \State $c \gets$ RunSize($runs[runs.size() - 3]$)
        \If{$a \geq b \textbf{ or } (a + b) \geq c$}
            \If{$a \geq c$}
                \State Merge $runs[runs.size() - 3]$ and $runs[runs.size() - 2]$
            \Else
                \State Merge $runs[runs.size() - 2]$ and $runs[runs.size() - 1]$
            \EndIf
        \ElseIf{$runs.size() == 2$}
            \State Merge $runs[0]$ and $runs[1]$
        \EndIf
    \EndWhile
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{Tournament Sort}
\begin{algorithmic}
\Procedure{TournamentSort}{$arr, n$}
    \State $size \gets 1$
    \While{$size < n$}
        \State $size \gets size \times 2$
    \EndWhile
    \State Initialize $tree$ of size $2 \times size$
    \For{$i \gets 0$ to $n - 1$}
        \State $tree[size + i] \gets arr[i]$
    \EndFor
    \For{$i \gets size - 1$ to $1$}
        \State $tree[i] \gets \min(tree[2 \times i], tree[2 \times i + 1])$
    \EndFor
    \For{$i \gets 0$ to $n - 1$}
        \State $arr[i] \gets tree[1]$
        \State $cur \gets size + i$
        \While{$cur \geq 1$}
            \State $tree[cur] \gets \textbf{INT\_MAX}$
            \State $cur \gets cur / 2$
            \State $tree[cur] \gets \min(tree[2 \times cur], tree[2 \times cur + 1])$
        \EndWhile
    \EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}
